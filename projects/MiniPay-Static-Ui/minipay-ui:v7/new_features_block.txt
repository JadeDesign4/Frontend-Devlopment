    const receiveModal = $('#receive-modal');

    function generateQR(text) {
        const size = 25;
        const cell = 7;
        let rects = '';
        for (let y = 0; y < size; y++) {
            for (let x = 0; x < size; x++) {
                const hash = Math.sin((x + 1) * (y + 7) * text.length) * 10000;
                if ((hash - Math.floor(hash)) > 0.45) {
                    rects += '<rect x=\'' + (x * cell) + '\' y=\'' + (y * cell) + '\' width=\'' + cell + '\' height=\'' + cell + '\' fill=\'#0f1214\'/>';
                }
            }
        }
        return '<svg viewBox=\'0 0 ' + (size * cell) + ' ' + (size * cell) + '\' xmlns=\'http://www.w3.org/2000/svg\' aria-label=\'QR code\'>' + rects + '</svg>';
    }

        if (qrCode) qrCode.innerHTML = generateQR(walletId);

    function generateSparkline(change) {
        const points = 12;
        const width = 200;
        const height = 40;
        const step = width / (points - 1);
        const trend = change >= 0 ? 1 : -1;
        let path = '';
        const yValues = [];
        for (let i = 0; i < points; i++) {
            const noise = Math.random() * 20 - 10;
            const y = height / 2 - (trend * (i / points) * 18) + noise;
            yValues.push(y);
            if (i === 0) {
                path += 'M ' + (i * step) + ' ' + y;
            } else {
                path += ' L ' + (i * step) + ' ' + y;
            }
        }
        const areaPath = path + ' L ' + width + ' ' + height + ' L 0 ' + height + ' Z';
        const strokeColor = trend >= 0 ? '#22c55e' : '#ef4444';
        return '<svg class=\'sparkline\' viewBox=\'0 0 ' + width + ' ' + height + '\' preserveAspectRatio=\'none\' xmlns=\'http://www.w3.org/2000/svg\'>\n            <path d=\'' + areaPath + '\' fill=\'' + strokeColor + '\' opacity=\'0.12\'/>\n            <path d=\'' + path + '\' fill=\'none\' stroke=\'' + strokeColor + '\' stroke-width=\'2.5\' stroke-linecap=\'round\' stroke-linejoin=\'round\'/>\n        </svg>';
    }

    function renderPortfolio() {
        if (!tokenList) return;
        const currency = currencySwitch?.dataset.currency || 'USD';
        const total = tokens.reduce((sum, t) => sum + t.balance * t.price, 0);
        const totalChange = tokens.reduce((sum, t) => sum + (t.balance * t.price / Math.max(total, 0.01)) * t.change, 0);

        if (portfolioValue) portfolioValue.textContent = formatCurrency(total, currency);
        if (portfolioChange) {
            portfolioChange.textContent = (totalChange >= 0 ? '+' : '') + totalChange.toFixed(2) + '%';
            portfolioChange.className = 'portfolio-change ' + (totalChange >= 0 ? 'up' : 'down');
        }

        tokenList.innerHTML = tokens.map(token => {
            const value = token.balance * token.price;
            const changeClass = token.change >= 0 ? 'up' : 'down';
            return `
                <article class='token-card'>
                    <div class='token-meta'>
                        <div class='token-name'>
                            <span class='token-symbol'>${escapeHtml(token.symbol)}</span>
                            <span class='token-full'>${escapeHtml(token.name)}</span>
                        </div>
                        <div class='token-balance'>
                            <span class='token-amount'>${escapeHtml(token.balance.toLocaleString('en-US', { maximumFractionDigits: 6 }))}</span>
                            <span class='token-value'>${formatCurrency(value, currency)}</span>
                        </div>
                    </div>
                    ${generateSparkline(token.change)}
                    <div class='token-change ${changeClass}'>${token.change >= 0 ? '+' : ''}${token.change.toFixed(1)}% (24h)</div>
                </article>
            `;
        }).join('');
    }
